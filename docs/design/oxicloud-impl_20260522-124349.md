```rust
use axum::{
    http::{header, StatusCode},
    response::{IntoResponse, Response},
};
use thiserror::Error;

const DAV_ALLOW: &str =
    "OPTIONS, HEAD, GET, PUT, DELETE, PROPFIND, PROPPATCH, MKCOL, MKCALENDAR, REPORT";

#[derive(Debug, Error)]
pub enum AppError {
    #[error("Bad request: {0}")]
    BadRequest(String),

    #[error("Unauthorized")]
    Unauthorized,

    #[error("Forbidden: {0}")]
    Forbidden(String),

    #[error("Not found")]
    NotFound,

    #[error("Method not allowed")]
    MethodNotAllowed,

    #[error("Conflict: {0}")]
    Conflict(String),

    #[error("Precondition failed: {0}")]
    PreconditionFailed(String),

    #[error("Unsupported media type: {0}")]
    UnsupportedMediaType(String),

    #[error("Database error: {0}")]
    Database(sqlx::Error),

    #[error("Internal server error")]
    Internal,
}

impl AppError {
    pub fn bad_request(message: impl Into<String>) -> Self {
        Self::BadRequest(message.into())
    }

    pub fn forbidden(message: impl Into<String>) -> Self {
        Self::Forbidden(message.into())
    }

    pub fn conflict(message: impl Into<String>) -> Self {
        Self::Conflict(message.into())
    }

    pub fn precondition_failed(message: impl Into<String>) -> Self {
        Self::PreconditionFailed(message.into())
    }

    pub fn unsupported_media_type(message: impl Into<String>) -> Self {
        Self::UnsupportedMediaType(message.into())
    }

    pub fn status_code(&self) -> StatusCode {
        match self {
            AppError::BadRequest(_) => StatusCode::BAD_REQUEST,
            AppError::Unauthorized => StatusCode::UNAUTHORIZED,
            AppError::Forbidden(_) => StatusCode::FORBIDDEN,
            AppError::NotFound => StatusCode::NOT_FOUND,
            AppError::MethodNotAllowed => StatusCode::METHOD_NOT_ALLOWED,
            AppError::Conflict(_) => StatusCode::CONFLICT,
            AppError::PreconditionFailed(_) => StatusCode::PRECONDITION_FAILED,
            AppError::UnsupportedMediaType(_) => StatusCode::UNSUPPORTED_MEDIA_TYPE,
            AppError::Database(_) | AppError::Internal => StatusCode::INTERNAL_SERVER_ERROR,
        }
    }
}

impl From<sqlx::Error> for AppError {
    fn from(error: sqlx::Error) -> Self {
        match error {
            sqlx::Error::RowNotFound => AppError::NotFound,
            other => AppError::Database(other),
        }
    }
}

impl IntoResponse for AppError {
    fn into_response(self) -> Response {
        let status = self.status_code();
        let body = self.to_string();

        match status {
            StatusCode::UNAUTHORIZED => (
                status,
                [(header::WWW_AUTHENTICATE, r#"Basic realm="OxiCloud""#)],
                body,
            )
                .into_response(),
            StatusCode::METHOD_NOT_ALLOWED => {
                (status, [(header::ALLOW, DAV_ALLOW)], body).into_response()
            }
            _ => (status, body).into_response(),
        }
    }
}
```

## `src/interfaces/api/handlers/webdav.rs`

```rust
use axum::{
    body::{to_bytes, Body},
    extract::{Request, State},
    http::{header, HeaderValue, Method, Response, StatusCode},
};
use std::sync::Arc;

use crate::{
    application::adapters::webdav::{
        build_error_xml, format_http_date, is_supported_object_content_type, multi_status,
        normalize_href, split_parent_href_and_name, DavDepth, PutObjectRequest, WebDavAdapter,
    },
    domain::error::AppError,
    state::AppState,
};

const OWNER_USERNAME: &str = "user1";
const MAX_DAV_BODY_BYTES: usize = 16 * 1024 * 1024;
const DAV_ALLOW: &str =
    "OPTIONS, HEAD, GET, PUT, DELETE, PROPFIND, PROPPATCH, MKCOL, MKCALENDAR, REPORT";
const DAV_HEADER: &str = "1, 3, extended-mkcol, calendar-access, addressbook, sync-collection";

pub async fn handle_webdav(
    State(state): State<Arc<AppState>>,
    request: Request<Body>,
) -> Result<Response<Body>, AppError> {
    let method = request.method().clone();
    let href = normalize_href(request.uri().path());

    if method == Method::GET && (href == "/.well-known/caldav" || href == "/.well-known/carddav") {
        return redirect_to_dav_root();
    }

    if should_redirect_collection_without_trailing_slash(&method, &href) {
        return redirect_to_collection_with_trailing_slash(&href, request.uri().query());
    }

    let adapter = WebDavAdapter::new(state.db_pool.clone());

    match method.as_str() {
        "OPTIONS" => options_response(),
        "PROPFIND" => handle_propfind(adapter, href, request).await,
        "REPORT" => handle_report(adapter, href, request).await,
        "PROPPATCH" => handle_proppatch(adapter, href, request).await,
        "GET" => handle_get(adapter, href, request, false).await,
        "HEAD" => handle_get(adapter, href, request, true).await,
        "PUT" => handle_put(adapter, href, request).await,
        "DELETE" => handle_delete(adapter, href, request).await,
        "MKCOL" => handle_mkcol(adapter, href, request).await,
        "MKCALENDAR" => handle_mkcalendar(adapter, href, request).await,
        "LOCK" | "UNLOCK" => method_not_allowed_response(),
        _ => method_not_allowed_response(),
    }
}

async fn handle_propfind(
    adapter: WebDavAdapter,
    href: String,
    request: Request<Body>,
) -> Result<Response<Body>, AppError> {
    let depth = depth_from_headers(request.headers().get("Depth"), DavDepth::Infinity);

    if depth == DavDepth::Infinity {
        return empty_response(StatusCode::FORBIDDEN);
    }

    let body = to_bytes(request.into_body(), MAX_DAV_BODY_BYTES)
        .await
        .map_err(|error| AppError::bad_request(format!("Failed to read request body: {error}")))?;

    let xml = adapter
        .propfind(OWNER_USERNAME, &href, depth, body.as_ref())
        .await?;

    xml_response(multi_status(), xml)
}

async fn handle_report(
    adapter: WebDavAdapter,
    href: String,
    request: Request<Body>,
) -> Result<Response<Body>, AppError> {
    let depth = depth_from_headers(request.headers().get("Depth"), DavDepth::Zero);

    if depth == DavDepth::Infinity {
        return empty_response(StatusCode::FORBIDDEN);
    }

    let body = to_bytes(request.into_body(), MAX_DAV_BODY_BYTES)
        .await
        .map_err(|error| AppError::bad_request(format!("Failed to read request body: {error}")))?;

    match adapter
        .report(OWNER_USERNAME, &href, depth, body.as_ref())
        .await
    {
        Ok(xml) => xml_response(multi_status(), xml),
        Err(AppError::Forbidden(message)) if message.contains("Invalid sync token") => {
            xml_response(StatusCode::FORBIDDEN, build_error_xml("valid-sync-token")?)
        }
        Err(AppError::Forbidden(_)) => {
            xml_response(StatusCode::FORBIDDEN, build_error_xml("supported-report")?)
        }
        Err(error) => Err(error),
    }
}

async fn handle_proppatch(
    adapter: WebDavAdapter,
    href: String,
    request: Request<Body>,
) -> Result<Response<Body>, AppError> {
    let body = to_bytes(request.into_body(), MAX_DAV_BODY_BYTES)
        .await
        .map_err(|error| AppError::bad_request(format!("Failed to read request body: {error}")))?;

    let xml = adapter
        .proppatch(OWNER_USERNAME, &href, body.as_ref())
        .await?;

    xml_response(multi_status(), xml)
}

async fn handle_get(
    adapter: WebDavAdapter,
    href: String,
    request: Request<Body>,
    head_only: bool,
) -> Result<Response<Body>, AppError> {
    let if_none_match = request
        .headers()
        .get(header::IF_NONE_MATCH)
        .and_then(|value| value.to_str().ok())
        .map(ToOwned::to_owned);

    let object = adapter
        .get_object(OWNER_USERNAME, &href, if_none_match.as_deref(), head_only)
        .await?;

    if object.not_modified {
        return Response::builder()
            .status(StatusCode::NOT_MODIFIED)
            .header(header::ETAG, object.etag)
            .header(header::CONTENT_LENGTH, "0")
            .body(Body::empty())
            .map_err(|_| AppError::Internal);
    }

    let builder = Response::builder()
        .status(StatusCode::OK)
        .header(header::CONTENT_TYPE, object.content_type)
        .header(header::ETAG, object.etag)
        .header(header::LAST_MODIFIED, format_http_date(object.last_modified))
        .header(header::CONTENT_LENGTH, object.content_length.to_string());

    if head_only {
        builder.body(Body::empty()).map_err(|_| AppError::Internal)
    } else {
        builder
            .body(Body::from(object.content))
            .map_err(|_| AppError::Internal)
    }
}

async fn handle_put(
    adapter: WebDavAdapter,
    href: String,
    request: Request<Body>,
) -> Result<Response<Body>, AppError> {
    let content_type = request
        .headers()
        .get(header::CONTENT_TYPE)
        .and_then(|value| value.to_str().ok())
        .unwrap_or("")
        .to_owned();

    if !is_supported_object_content_type(&content_type) {
        return empty_response(StatusCode::UNSUPPORTED_MEDIA_TYPE);
    }

    let if_match = request
        .headers()
        .get(header::IF_MATCH)
        .and_then(|value| value.to_str().ok())
        .map(ToOwned::to_owned);

    let if_none_match = request
        .headers()
        .get(header::IF_NONE_MATCH)
        .and_then(|value| value.to_str().ok())
        .map(ToOwned::to_owned);

    let (normalized, parent_href, name) = split_parent_href_and_name(&href, false)?;

    let body = to_bytes(request.into_body(), MAX_DAV_BODY_BYTES)
        .await
        .map_err(|error| AppError::bad_request(format!("Failed to read request body: {error}")))?;

    let response = adapter
        .put_object(PutObjectRequest {
            owner_username: OWNER_USERNAME.to_owned(),
            href: normalized.clone(),
            parent_href,
            name,
            content_type,
            content: body.to_vec(),
            if_match,
            if_none_match,
        })
        .await?;

    let status = if response.created {
        StatusCode::CREATED
    } else {
        StatusCode::NO_CONTENT
    };

    let mut builder = Response::builder()
        .status(status)
        .header(header::ETAG, response.etag);

    if response.created {
        builder = builder.header(header::LOCATION, normalized);
    }

    builder
        .header(header::CONTENT_LENGTH, "0")
        .body(Body::empty())
        .map_err(|_| AppError::Internal)
}

async fn handle_delete(
    adapter: WebDavAdapter,
    href: String,
    request: Request<Body>,
) -> Result<Response<Body>, AppError> {
    let if_match = request
        .headers()
        .get(header::IF_MATCH)
        .and_then(|value| value.to_str().ok())
        .map(ToOwned::to_owned);

    adapter
        .delete(OWNER_USERNAME, &href, if_match.as_deref())
        .await?;

    empty_response(StatusCode::NO_CONTENT)
}

async fn handle_mkcol(
    adapter: WebDavAdapter,
    href: String,
    request: Request<Body>,
) -> Result<Response<Body>, AppError> {
    let content_type = request
        .headers()
        .get(header::CONTENT_TYPE)
        .and_then(|value| value.to_str().ok())
        .unwrap_or("")
        .to_ascii_lowercase();

    let body = to_bytes(request.into_body(), MAX_DAV_BODY_BYTES)
        .await
        .map_err(|error| AppError::bad_request(format!("Failed to read request body: {error}")))?;

    if !body.is_empty()
        && !content_type.starts_with("application/xml")
        && !content_type.starts_with("text/xml")
    {
        return empty_response(StatusCode::UNSUPPORTED_MEDIA_TYPE);
    }

    adapter.mkcol(OWNER_USERNAME, &href, body.as_ref()).await?;
    empty_response(StatusCode::CREATED)
}

async fn handle_mkcalendar(

) -> Result<Response<Body>, AppError> {
    let content_type = request

}

fn options_response() -> Result<Response<Body>, AppError> {
    Response::builder()
        .status(StatusCode::OK)
        .header("DAV", DAV_HEADER)
        .header("MS-Author-Via", "DAV")
        .header(header::ALLOW, DAV_ALLOW)
        .header(header::CONTENT_LENGTH, "0")
        .body(Body::empty())
        .map_err(|_| AppError::Internal)
}

fn redirect_to_dav_root() -> Result<Response<Body>, AppError> {
    Response::builder()
        .status(StatusCode::MOVED_PERMANENTLY)
        .header(header::LOCATION, "/dav/")
        .header(header::CONTENT_LENGTH, "0")
        .body(Body::empty())
        .map_err(|_| AppError::Internal)
}

fn redirect_to_collection_with_trailing_slash(
    href: &str,
    query: Option<&str>,
) -> Result<Response<Body>, AppError> {
    let mut location = format!("{href}/");

    if let Some(query) = query {
        if !query.is_empty() {
            location.push('?');
            location.push_str(query);
        }
    }

    Response::builder()
        .status(StatusCode::MOVED_PERMANENTLY)
        .header(header::LOCATION, location)
        .header(header::CONTENT_LENGTH, "0")
        .body(Body::empty())
        .map_err(|_| AppError::Internal)
}

fn method_not_allowed_response() -> Result<Response<Body>, AppError> {
    Response::builder()
        .status(StatusCode::METHOD_NOT_ALLOWED)
        .header(header::ALLOW, DAV_ALLOW)
        .header(header::CONTENT_LENGTH, "0")
        .body(Body::empty())
        .map_err(|_| AppError::Internal)
}

fn xml_response(status: StatusCode, xml: String) -> Result<Response<Body>, AppError> {
    Response::builder()
        .status(status)
        .header(header::CONTENT_TYPE, "application/xml; charset=utf-8")
        .body(Body::from(xml))
        .map_err(|_| AppError::Internal)
}

fn empty_response(status: StatusCode) -> Result<Response<Body>, AppError> {
    Response::builder()
        .status(status)
        .header(header::CONTENT_LENGTH, "0")
        .body(Body::empty())
        .map_err(|_| AppError::Internal)
}

fn depth_from_headers(value: Option<&HeaderValue>, default: DavDepth) -> DavDepth {
    let Some(value) = value.and_then(|value| value.to_str().ok()) else {
        return default;
    };

    match value.trim().to_ascii_lowercase().as_str() {
        "0" => DavDepth::Zero,
        "1" => DavDepth::One,
        "infinity" => DavDepth::Infinity,
        _ => default,
    }
}

fn should_redirect_collection_without_trailing_slash(method: &Method, href: &str) -> bool {
    if href == "/" || href.ends_with('/') {
        return false;
    }

    if !matches!(
        method.as_str(),
        "OPTIONS" | "HEAD" | "GET" | "PROPFIND" | "REPORT"
    ) {
        return false;
    }

    let parts = href
        .trim_matches('/')
        .split('/')
        .filter(|part| !part.is_empty())
        .collect::<Vec<_>>();

    match parts.as_slice() {
        ["dav"] => true,
        ["dav", "principals"] => true,
        ["dav", "principals", _] => true,
        ["dav", "calendars"] => true,
        ["dav", "calendars", _] => true,
        ["dav", "calendars", _, collection] => !collection.contains('.'),
        ["dav", "addressbooks"] => true,
        ["dav", "addressbooks", _] => true,
        ["dav", "addressbooks", _, collection] => !collection.contains('.'),
        _ => false,
    }
}
```

```rust
#[derive(Debug, Clone, sqlx::FromRow)]
struct DavNode {
    id: Uuid,
    parent_id: Option<Uuid>,
    name: String,
    is_directory: bool,
    size: i64,
    content_type: Option<String>,
    created_at: DateTime<Utc>,
    updated_at: DateTime<Utc>,

impl DavNode {
    fn is_collection(&self) -> bool {
        matches!(
            self.dav_resource_type.as_str(),
            "collection" | "principal" | "calendar_home" | "addressbook_home" | "calendar" | "addressbook"
        )
    }

    fn is_object(&self) -> bool {
        matches!(
            self.dav_resource_type.as_str(),
            "calendar_object" | "address_object" | "file"
        )
    }

    fn is_calendar_collection(&self) -> bool {
        self.dav_resource_type == "calendar"
    }

    fn is_addressbook_collection(&self) -> bool {
        self.dav_resource_type == "addressbook"
    }

    fn is_calendar_object(&self) -> bool {
        self.dav_resource_type == "calendar_object"
    }

    fn is_address_object(&self) -> bool {
        self.dav_resource_type == "address_object"
    }

    fn quoted_etag(&self) -> String {
        quote_etag(&self.etag)
    }

    fn effective_content_type(&self) -> String {
        self.content_type.clone().unwrap_or_else(|| {
            if self.is_calendar_object() {
                "text/calendar; charset=utf-8".to_owned()
            } else if self.is_address_object() {
                "text/vcard; charset=utf-8".to_owned()
            } else if self.is_collection() {
                "httpd/unix-directory".to_owned()
            } else {
                "application/octet-stream".to_owned()
            }
        })
    }
}

impl WebDavAdapter {
    pub fn new(db_pool: PgPool) -> Self {
        Self { db_pool }
    }

    pub async fn propfind(
        &self,
        owner_username: &str,
        href: &str,
        depth: DavDepth,
        body: &[u8],
    ) -> Result<String, AppError> {
        if depth == DavDepth::Infinity {
            return Err(AppError::forbidden("Depth infinity is not supported"));
        }

        let parsed = parse_propfind_body(body)?;
        let target = self
            .find_active(owner_username, &normalize_href(href))
            .await?
            .ok_or(AppError::NotFound)?;

        let mut nodes = vec![target.clone()];

        if depth == DavDepth::One && target.is_collection() {
            nodes.extend(self.list_children(target.id).await?);
        }

        let responses = nodes
            .iter()
            .map(|node| {
                let properties = match parsed.mode {
                    PropfindMode::Allprop | PropfindMode::Propname => live_properties(node),
                    PropfindMode::Prop => parsed.properties.clone(),
                };

                response_for_node(node, &properties, parsed.mode)
            })
            .collect::<Vec<_>>();

        build_multistatus_xml(&Multistatus {
            responses,
            sync_token: None,
        })
    }

    pub async fn report(
        &self,
        owner_username: &str,
        href: &str,
        depth: DavDepth,
        body: &[u8],
    ) -> Result<String, AppError> {
        if depth == DavDepth::Infinity {
            return Err(AppError::forbidden("Depth infinity is not supported for REPORT"));
        }

        let report = parse_report_body(body)?;
        let target = self
            .find_active(owner_username, &normalize_href(href))
            .await?
            .ok_or(AppError::NotFound)?;

        let multistatus = match report.kind {
            ReportKind::CalendarQuery => {
                if !target.is_calendar_collection() {
                    return Err(AppError::forbidden(
                        "calendar-query is only supported on calendar collections",
                    ));
                }

                let objects = self
                    .list_children_by_type(target.id, "calendar_object")
                    .await?;

                Multistatus {
                    responses: objects
                        .iter()
                        .map(|node| {
                            ResponseItem::propstats(
                                node.dav_path.clone(),
                                vec![Propstat::ok(render_requested_properties(
                                    node,
                                    &report.properties,
                                ))],
                            )
                        })
                        .collect(),
                    sync_token: None,
                }
            }
            ReportKind::AddressbookQuery => {
                if !target.is_addressbook_collection() {
                    return Err(AppError::forbidden(
                        "addressbook-query is only supported on address book collections",
                    ));
                }

                let objects = self
                    .list_children_by_type(target.id, "address_object")
                    .await?;

                Multistatus {
                    responses: objects
                        .iter()
                        .map(|node| {
                            ResponseItem::propstats(
                                node.dav_path.clone(),
                                vec![Propstat::ok(render_requested_properties(
                                    node,
                                    &report.properties,
                                ))],
                            )
                        })
                        .collect(),
                    sync_token: None,
                }
            }
            ReportKind::CalendarMultiget | ReportKind::AddressbookMultiget => {
                let found = self
                    .list_objects_by_hrefs(owner_username, &report.hrefs)
                    .await?;

                let found_by_href = found
                    .into_iter()
                    .map(|node| (node.dav_path.clone(), node))
                    .collect::<HashMap<_, _>>();

                Multistatus {
                    responses: report
                        .hrefs
                        .iter()
                        .map(|href| {
                            let href = normalize_href(href);

                            if let Some(node) = found_by_href.get(&href) {
                                ResponseItem::propstats(
                                    href,
                                    vec![Propstat::ok(render_requested_properties(
                                        node,
                                        &report.properties,
                                    ))],
                                )
                            } else {
                                ResponseItem::status(href, StatusCode::NOT_FOUND)
                            }
                        })
                        .collect(),
                    sync_token: None,
                }
            }
            ReportKind::SyncCollection => {
                if !target.is_collection() {
                    return Err(AppError::forbidden(
                        "sync-collection is only supported on collections",
                    ));
                }

                let after_version = report
                    .sync_token
                    .as_deref()
                    .and_then(parse_sync_token_version)
                    .unwrap_or(0);

                if after_version > target.sync_version {
                    return Err(AppError::forbidden("Invalid sync token"));
                }

                let changes = self.list_sync_changes(target.id, after_version).await?;

                Multistatus {
                    responses: changes
                        .into_iter()
                        .map(|change| {
                            if change.change_kind == "deleted" {
                                ResponseItem::status(change.href, StatusCode::NOT_FOUND)
                            } else {
                                let mut properties = Vec::new();

                                for requested in &report.properties {
                                    if requested.namespace == NS_DAV
                                        && requested.local_name == "getetag"
                                    {
                                        if let Some(etag) = change.etag.as_deref() {
                                            properties.push(PropertyValue::text(
                                                requested.clone(),
                                                quote_etag(etag),
                                            ));
                                        }
                                    }
                                }

                                ResponseItem::propstats(change.href, vec![Propstat::ok(properties)])
                            }
                        })
                        .collect(),
                    sync_token: Some(sync_token(&target.dav_path, target.sync_version)),
                }
            }
        };

        build_multistatus_xml(&multistatus)
    }

    pub async fn proppatch(
        &self,
        owner_username: &str,
        href: &str,
        body: &[u8],
    ) -> Result<String, AppError> {
        let href = normalize_href(href);
        let patches = parse_proppatch_body(body)?;
        let node = self
            .find_active(owner_username, &href)
            .await?
            .ok_or(AppError::NotFound)?;

        let mut ok = Vec::new();
        let mut forbidden = Vec::new();

        let mut display_name: Option<String> = None;
        let mut calendar_description: Option<String> = None;
        let mut calendar_timezone: Option<String> = None;
        let mut calendar_color: Option<String> = None;
        let mut calendar_order: Option<i32> = None;
        let mut addressbook_description: Option<String> = None;

        for patch in patches {
            if is_writable_property(&patch.property) {
                let value = patch.value.unwrap_or_default();

                match (patch.property.namespace.as_str(), patch.property.local_name.as_str()) {
                    (NS_DAV, "displayname") => display_name = Some(value),
                    (NS_CALDAV, "calendar-description") => calendar_description = Some(value),
                    (NS_CALDAV, "calendar-timezone") => calendar_timezone = Some(value),
                    (NS_APPLE_ICAL, "calendar-color") => calendar_color = Some(value),
                    (NS_APPLE_ICAL, "calendar-order") => {
                        calendar_order = Some(value.parse::<i32>().unwrap_or(1).max(0));
                    }
                    (NS_CARDDAV, "addressbook-description") => {
                        addressbook_description = Some(value);
                    }
                    _ => {}
                }

                ok.push(PropertyValue::empty(patch.property));
            } else {
                forbidden.push(PropertyValue::empty(patch.property));
            }
        }

        if !ok.is_empty() {
            sqlx::query(
                r#"
                UPDATE nodes
                SET display_name = COALESCE($2, display_name),
                    calendar_description = COALESCE($3, calendar_description),
                    calendar_timezone = COALESCE($4, calendar_timezone),
                    calendar_color = COALESCE($5, calendar_color),
                    calendar_order = COALESCE($6, calendar_order),
                    addressbook_description = COALESCE($7, addressbook_description),
                    sync_version = sync_version + 1
                WHERE id = $1
                "#,
            )
            .bind(node.id)
            .bind(display_name)
            .bind(calendar_description)
            .bind(calendar_timezone)
            .bind(calendar_color)
            .bind(calendar_order)
            .bind(addressbook_description)
            .execute(&self.db_pool)
            .await?;
        }

        let mut propstats = Vec::new();

        if !ok.is_empty() {
            propstats.push(Propstat::ok(ok));
        }

        if !forbidden.is_empty() {
            propstats.push(Propstat::new(forbidden, StatusCode::FORBIDDEN));
        }

        build_multistatus_xml(&Multistatus {
            responses: vec![ResponseItem::propstats(href, propstats)],
            sync_token: None,
        })
    }

    pub async fn get_object(
        &self,
        owner_username: &str,
        href: &str,
        if_none_match: Option<&str>,
        head_only: bool,
    ) -> Result<GetObjectResponse, AppError> {
        let node = self
            .find_active(owner_username, &normalize_href(href))
            .await?
            .ok_or(AppError::NotFound)?;

        if !node.is_object() {
            return Err(AppError::forbidden("GET is only supported for object resources"));
        }

        let etag = node.quoted_etag();
        let not_modified = if_none_match
            .map(|candidate| candidate.trim() == etag)
            .unwrap_or(false);

        Ok(GetObjectResponse {
            content: if head_only || not_modified {
                Vec::new()
            } else {
                node.content
            },
            content_type: node.effective_content_type(),
            content_length: node.size,
            etag,
            last_modified: node.updated_at,
            not_modified,
        })
    }

    pub async fn put_object(
        &self,
        request: PutObjectRequest,
    ) -> Result<PutObjectResponse, AppError> {
        if !is_supported_object_content_type(&request.content_type) {
            return Err(AppError::unsupported_media_type(format!(
                "Unsupported content type {}",
                request.content_type
            )));
        }

        let href = normalize_href(&request.href);
        let parent_href = normalize_collection_href(&request.parent_href);

        let parent = self
            .find_active(&request.owner_username, &parent_href)
            .await?
            .ok_or_else(|| AppError::conflict("Parent collection does not exist"))?;

        if !parent.is_collection() {
            return Err(AppError::conflict("Parent is not a collection"));
        }

        let resource_type = if request.href.to_ascii_lowercase().ends_with(".ics")
            || request
                .content_type
                .to_ascii_lowercase()
                .starts_with("text/calendar")
        {
            if !parent.is_calendar_collection() {
                return Err(AppError::unsupported_media_type(
                    "Calendar objects must be stored in calendar collections",
                ));
            }

            "calendar_object"
        } else if request.href.to_ascii_lowercase().ends_with(".vcf")
            || request
                .content_type
                .to_ascii_lowercase()
                .starts_with("text/vcard")
        {
            if !parent.is_addressbook_collection() {
                return Err(AppError::unsupported_media_type(
                    "Address objects must be stored in address book collections",
                ));
            }

            "address_object"
        } else {
            "file"
        };

        let existing = self.find_active(&request.owner_username, &href).await?;

        if request.if_none_match.as_deref().map(str::trim) == Some("*") && existing.is_some() {
            return Err(AppError::precondition_failed(
                "If-None-Match precondition failed",
            ));
        }

        if let Some(if_match) = request.if_match.as_deref() {
            let Some(existing_node) = existing.as_ref() else {
                return Err(AppError::precondition_failed(
                    "If-Match precondition failed: resource does not exist",
                ));
            };

            if existing_node.quoted_etag() != if_match.trim() {
                return Err(AppError::precondition_failed(
                    "If-Match precondition failed: ETag mismatch",
                ));
            }
        }

        let created = existing.is_none();
        let next_version = parent.sync_version + 1;
        let etag = new_raw_etag(&href);
        let content_type = normalized_object_content_type(&request.content_type);
        let size = request.content.len() as i64;

        let sql = format!("SELECT {NODE_COLUMNS} FROM nodes WHERE id = $1");

        let node = if let Some(existing_node) = existing {
            sqlx::query(
                r#"
                UPDATE nodes
                SET content = $2,
                    size = $3,
                    content_type = $4,
                    etag = $5,
                    sync_version = $6,
                    deleted_at = NULL
                WHERE id = $1
                "#,
            )
            .bind(existing_node.id)
            .bind(request.content)
            .bind(size)
            .bind(content_type)
            .bind(&etag)
            .bind(next_version)
            .execute(&self.db_pool)
            .await?;

            sqlx::query_as::<_, DavNode>(&sql)
                .bind(existing_node.id)
                .fetch_one(&self.db_pool)
                .await?
        } else {
            let insert_sql = format!(
                r#"
                INSERT INTO nodes (
                    parent_id,
                    name,
                    is_directory,
                    size,
                    content_type,
                    owner_username,
                    dav_path,
                    dav_resource_type,
                    display_name,
                    content,
                    etag,
                    sync_version
                )
                VALUES ($1, $2, FALSE, $3, $4, $5, $6, $7, $8, $9, $10, $11)
                RETURNING {NODE_COLUMNS}
                "#
            );

            sqlx::query_as::<_, DavNode>(&insert_sql)
                .bind(parent.id)
                .bind(&request.name)
                .bind(size)
                .bind(content_type)
                .bind(&request.owner_username)
                .bind(&href)
                .bind(resource_type)
                .bind(&request.name)
                .bind(request.content)
                .bind(&etag)
                .bind(next_version)
                .fetch_one(&self.db_pool)
                .await?
        };

        self.update_collection_sync(parent.id, next_version).await?;

        self.record_sync_change(
            parent.id,
            Some(node.id),
            &href,
            next_version,
            if created { "created" } else { "updated" },
            Some(&node.etag),
            node.content_type.as_deref(),
        )
        .await?;

        Ok(PutObjectResponse {
            created,
            etag: node.quoted_etag(),
        })
    }

    pub async fn delete(
        &self,
        owner_username: &str,
        href: &str,
        if_match: Option<&str>,
    ) -> Result<(), AppError> {
        let href = normalize_href(href);
        let node = self
            .find_active(owner_username, &href)
            .await?
            .ok_or(AppError::NotFound)?;

        if let Some(if_match) = if_match {
            if node.quoted_etag() != if_match.trim() {
                return Err(AppError::precondition_failed(
                    "If-Match precondition failed: ETag mismatch",
                ));
            }
        }

        if node.is_collection() && self.active_child_count(node.id).await? > 0 {
            return Err(AppError::forbidden(
                "Deleting non-empty collections is not supported",
            ));
        }

        let collection = if node.is_collection() {
            node.clone()
        } else {
            let parent_id = node
                .parent_id
                .ok_or_else(|| AppError::conflict("Object has no parent collection"))?;

            self.find_by_id(parent_id).await?.ok_or(AppError::NotFound)?
        };

        let next_version = collection.sync_version + 1;

        sqlx::query(
            r#"
            UPDATE nodes
            SET deleted_at = NOW(),
                sync_version = $2
            WHERE id = $1
              AND deleted_at IS NULL
            "#,
        )
        .bind(node.id)
        .bind(next_version)
        .execute(&self.db_pool)
        .await?;

        self.update_collection_sync(collection.id, next_version).await?;

        self.record_sync_change(
            collection.id,
            Some(node.id),
            &node.dav_path,
            next_version,
            "deleted",
            None,
            node.content_type.as_deref(),
        )
        .await?;

        Ok(())
    }

            SELECT id,
                   collection_id,
                   node_id,
                   href,
                   change_version,
                   change_kind,
                   etag,
                   content_type,
                   changed_at
            FROM dav_sync_changes
            WHERE collection_id = $1
              AND change_version > $2
            ORDER BY change_version ASC, id ASC
            "#,
        )
        .bind(collection_id)
        .bind(after_version)
        .fetch_all(&self.db_pool)
        .await?)
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct PropertyName {
    namespace: String,
    local_name: String,
}

impl PropertyName {
    fn new(namespace: impl Into<String>, local_name: impl Into<String>) -> Self {
        Self {
            namespace: namespace.into(),
            local_name: local_name.into(),
        }
    }

    fn dav(local_name: &str) -> Self {
        Self::new(NS_DAV, local_name)
    }

    fn caldav(local_name: &str) -> Self {
        Self::new(NS_CALDAV, local_name)
    }

    fn carddav(local_name: &str) -> Self {
        Self::new(NS_CARDDAV, local_name)
    }

    fn cs(local_name: &str) -> Self {
        Self::new(NS_CALENDAR_SERVER, local_name)
    }

    fn ical(local_name: &str) -> Self {
        Self::new(NS_APPLE_ICAL, local_name)
    }
}

#[derive(Debug, Clone)]

    Empty,
    Text(String),
    Href(String),
    ResourceType,
    SupportedCalendarComponents(Vec<String>),
    SupportedAddressData(Vec<String>),
    SupportedReportSet(Vec<PropertyName>),

#[derive(Debug, Clone)]

    name: PropertyName,
    value: PropertyRenderedValue,
}

impl PropertyValue {
    fn empty(name: PropertyName) -> Self {
        Self {
            name,
            value: PropertyRenderedValue::Empty,
        }
    }

    fn text(name: PropertyName, value: impl Into<String>) -> Self {
        Self {
            name,
            value: PropertyRenderedValue::Text(value.into()),
        }
    }
}

#[derive(Debug, Clone)]

    fn ok(properties: Vec<PropertyValue>) -> Self {
        Self::new(properties, StatusCode::OK)
    }

    fn not_found(properties: Vec<PropertyValue>) -> Self {
        Self::new(properties, StatusCode::NOT_FOUND)
    }

    fn new(properties: Vec<PropertyValue>, status: StatusCode) -> Self {
        Self { properties, status }
    }
}

fn response_for_node(
    node: &DavNode,
    requested: &[PropertyName],
    mode: PropfindMode,
) -> ResponseItem {
    let mut ok = Vec::new();
    let mut missing = Vec::new();

    for property in requested {
        if mode == PropfindMode::Propname {
            ok.push(PropertyValue::empty(property.clone()));
            continue;
        }

        match render_property(node, property) {
            Some(value) => ok.push(value),
            None => missing.push(PropertyValue::empty(property.clone())),
        }
    }

    let mut propstats = Vec::new();

    if !ok.is_empty() {
        propstats.push(Propstat::ok(ok));
    }

    if !missing.is_empty() {
        propstats.push(Propstat::not_found(missing));
    }

    ResponseItem::propstats(node.dav_path.clone(), propstats)
}

fn render_requested_properties(node: &DavNode, requested: &[PropertyName]) -> Vec<PropertyValue> {
    requested
        .iter()
        .filter_map(|property| render_property(node, property))
        .collect()
}

fn render_property(node: &DavNode, property: &PropertyName) -> Option<PropertyValue> {
    let value = match (property.namespace.as_str(), property.local_name.as_str()) {
        (NS_DAV, "displayname") => PropertyRenderedValue::Text(node.display_name.clone()),
        (NS_DAV, "resourcetype") => PropertyRenderedValue::ResourceType,
        (NS_DAV, "getcontentlength") if node.is_object() => {
            PropertyRenderedValue::Text(node.size.to_string())
        }
        (NS_DAV, "getcontenttype") if node.is_object() => {
            PropertyRenderedValue::Text(node.effective_content_type())
        }
        (NS_DAV, "getetag") if node.is_object() => {
            PropertyRenderedValue::Text(node.quoted_etag())
        }
        (NS_DAV, "getlastmodified") if node.is_object() => {
            PropertyRenderedValue::Text(format_http_date(node.updated_at))
        }
        (NS_DAV, "creationdate") => PropertyRenderedValue::Text(node.created_at.to_rfc3339()),
        (NS_DAV, "current-user-principal") if node.dav_path == "/dav/" => {
            PropertyRenderedValue::Href(format!("/dav/principals/{}/", node.owner_username))
        }
        (NS_DAV, "principal-URL") if node.dav_resource_type == "principal" => {
            PropertyRenderedValue::Href(node.dav_path.clone())
        }
        (NS_DAV, "sync-token") if node.is_calendar_collection() || node.is_addressbook_collection() => {
            PropertyRenderedValue::Text(sync_token(&node.dav_path, node.sync_version))
        }
        (NS_DAV, "supported-report-set") if node.is_calendar_collection() => {
            PropertyRenderedValue::SupportedReportSet(vec![
                PropertyName::caldav("calendar-query"),
                PropertyName::caldav("calendar-multiget"),
                PropertyName::dav("sync-collection"),
            ])
        }
        (NS_DAV, "supported-report-set") if node.is_addressbook_collection() => {
            PropertyRenderedValue::SupportedReportSet(vec![
                PropertyName::carddav("addressbook-query"),
                PropertyName::carddav("addressbook-multiget"),
                PropertyName::dav("sync-collection"),
            ])
        }
        (NS_CALDAV, "calendar-home-set") if node.dav_resource_type == "principal" => {
            PropertyRenderedValue::Href(format!("/dav/calendars/{}/", node.owner_username))
        }
        (NS_CALDAV, "calendar-description") if node.is_calendar_collection() => {
            PropertyRenderedValue::Text(node.calendar_description.clone())
        }
        (NS_CALDAV, "calendar-timezone") if node.is_calendar_collection() => {
            PropertyRenderedValue::Text(node.calendar_timezone.clone())
        }
        (NS_CALDAV, "supported-calendar-component-set") if node.is_calendar_collection() => {
            PropertyRenderedValue::SupportedCalendarComponents(
                node.supported_calendar_components.clone(),
            )
        }
        (NS_CALDAV, "calendar-data") if node.is_calendar_object() => {
            PropertyRenderedValue::Text(String::from_utf8_lossy(&node.content).to_string())
        }
        (NS_CARDDAV, "addressbook-home-set") if node.dav_resource_type == "principal" => {
            PropertyRenderedValue::Href(format!("/dav/addressbooks/{}/", node.owner_username))
        }
        (NS_CARDDAV, "addressbook-description") if node.is_addressbook_collection() => {
            PropertyRenderedValue::Text(node.addressbook_description.clone())
        }
        (NS_CARDDAV, "supported-address-data") if node.is_addressbook_collection() => {
            PropertyRenderedValue::SupportedAddressData(
                node.supported_address_data_versions.clone(),
            )
        }
        (NS_CARDDAV, "address-data") if node.is_address_object() => {
            PropertyRenderedValue::Text(String::from_utf8_lossy(&node.content).to_string())
        }
        (NS_CALENDAR_SERVER, "getctag")
            if node.is_calendar_collection() || node.is_addressbook_collection() =>
        {
            PropertyRenderedValue::Text(node.sync_version.to_string())
        }
        (NS_APPLE_ICAL, "calendar-color") if node.is_calendar_collection() => {
            PropertyRenderedValue::Text(node.calendar_color.clone())
        }
        (NS_APPLE_ICAL, "calendar-order") if node.is_calendar_collection() => {
            PropertyRenderedValue::Text(node.calendar_order.to_string())
        }
        _ => return None,
    };

    Some(PropertyValue {
        name: property.clone(),
        value,
    })
}

fn live_properties(node: &DavNode) -> Vec<PropertyName> {
    let mut properties = vec![
        PropertyName::dav("displayname"),
        PropertyName::dav("resourcetype"),
        PropertyName::dav("creationdate"),
    ];

    if node.is_object() {
        properties.extend([
            PropertyName::dav("getcontentlength"),
            PropertyName::dav("getcontenttype"),
            PropertyName::dav("getetag"),
            PropertyName::dav("getlastmodified"),
        ]);
    }

    match node.dav_resource_type.as_str() {
        "collection" if node.dav_path == "/dav/" => {
            properties.push(PropertyName::dav("current-user-principal"));
        }
        "principal" => {
            properties.push(PropertyName::dav("principal-URL"));
            properties.push(PropertyName::caldav("calendar-home-set"));
            properties.push(PropertyName::carddav("addressbook-home-set"));
        }
        "calendar" => {
            properties.push(PropertyName::dav("sync-token"));
            properties.push(PropertyName::dav("supported-report-set"));
            properties.push(PropertyName::cs("getctag"));
            properties.push(PropertyName::ical("calendar-color"));
            properties.push(PropertyName::ical("calendar-order"));
            properties.push(PropertyName::caldav("calendar-description"));
            properties.push(PropertyName::caldav("calendar-timezone"));
            properties.push(PropertyName::caldav("supported-calendar-component-set"));
        }
        "addressbook" => {
            properties.push(PropertyName::dav("sync-token"));
            properties.push(PropertyName::dav("supported-report-set"));
            properties.push(PropertyName::cs("getctag"));
            properties.push(PropertyName::carddav("addressbook-description"));
            properties.push(PropertyName::carddav("supported-address-data"));
        }
        _ => {}
    }

    properties
}

pub fn parse_propfind_body(body: &[u8]) -> Result<ParsedPropfind, AppError> {
    if body.is_empty() {
        return Ok(ParsedPropfind {
            mode: PropfindMode::Allprop,
            properties: Vec::new(),
        });
    }

    let mut reader = Reader::from_reader(Cursor::new(body));
    reader.config_mut().trim_text(true);

    let mut buf = Vec::new();
    let mut namespaces = NamespaceScope::default();
    let mut mode = PropfindMode::Prop;
    let mut properties = Vec::new();
    let mut inside_prop = false;
    let mut prop_depth = 0usize;

    loop {
        match reader
            .read_event_into(&mut buf)
            .map_err(|error| AppError::bad_request(format!("Invalid XML: {error}")))?
        {
            Event::Start(event) => {
                namespaces.capture(&event);
                let (namespace, local) = namespaces.resolve(event.name().as_ref());

                if namespace == NS_DAV && local == "allprop" {
                    mode = PropfindMode::Allprop;
                } else if namespace == NS_DAV && local == "propname" {
                    mode = PropfindMode::Propname;
                } else if namespace == NS_DAV && local == "prop" {
                    inside_prop = true;
                    prop_depth = 0;
                    mode = PropfindMode::Prop;
                } else if inside_prop {
                    if prop_depth == 0 {
                        properties.push(PropertyName::new(namespace, local));
                    }
                    prop_depth += 1;
                }
            }
            Event::Empty(event) => {
                namespaces.capture(&event);
                let (namespace, local) = namespaces.resolve(event.name().as_ref());

                if namespace == NS_DAV && local == "allprop" {
                    mode = PropfindMode::Allprop;
                } else if namespace == NS_DAV && local == "propname" {
                    mode = PropfindMode::Propname;
                } else if inside_prop && prop_depth == 0 {
                    properties.push(PropertyName::new(namespace, local));
                }
            }
            Event::End(event) => {
                let (namespace, local) = namespaces.resolve(event.name().as_ref());

                if namespace == NS_DAV && local == "prop" {
                    inside_prop = false;
                    prop_depth = 0;
                } else if inside_prop && prop_depth > 0 {
                    prop_depth -= 1;
                }
            }
            Event::Eof => break,
            _ => {}
        }

        buf.clear();
    }

    Ok(ParsedPropfind { mode, properties })
}

fn parse_report_body(body: &[u8]) -> Result<ParsedReport, AppError> {
    if body.is_empty() {
        return Err(AppError::bad_request("REPORT requires XML body"));
    }

    let mut reader = Reader::from_reader(Cursor::new(body));
    reader.config_mut().trim_text(true);

    let mut buf = Vec::new();
    let mut namespaces = NamespaceScope::default();
    let mut root: Option<PropertyName> = None;
    let mut properties = Vec::new();
    let mut hrefs = Vec::new();
    let mut sync_token = None;
    let mut inside_prop = false;
    let mut prop_depth = 0usize;
    let mut text_target: Option<&'static str> = None;

    loop {
        match reader
            .read_event_into(&mut buf)
            .map_err(|error| AppError::bad_request(format!("Invalid XML: {error}")))?
        {
            Event::Start(event) => {
                namespaces.capture(&event);
                let (namespace, local) = namespaces.resolve(event.name().as_ref());

                if root.is_none() {
                    root = Some(PropertyName::new(namespace.clone(), local.clone()));
                }

                if namespace == NS_DAV && local == "prop" {
                    inside_prop = true;
                    prop_depth = 0;
                } else if namespace == NS_DAV && local == "href" {
                    text_target = Some("href");
                } else if namespace == NS_DAV && local == "sync-token" {
                    text_target = Some("sync-token");
                } else if inside_prop {
                    if prop_depth == 0 {
                        properties.push(PropertyName::new(namespace, local));
                    }
                    prop_depth += 1;
                }
            }
            Event::Empty(event) => {
                namespaces.capture(&event);
                let (namespace, local) = namespaces.resolve(event.name().as_ref());

                if root.is_none() {
                    root = Some(PropertyName::new(namespace.clone(), local.clone()));
                }

                if inside_prop && prop_depth == 0 {
                    properties.push(PropertyName::new(namespace, local));
                }
            }
            Event::Text(event) => {
                if let Some(target) = text_target {
                    let text = decode_xml_text(&event)?;

                    match target {
                        "href" => hrefs.push(normalize_href(&text)),
                        "sync-token" => sync_token = Some(text),
                        _ => {}
                    }
                }
            }
            Event::End(event) => {
                let (namespace, local) = namespaces.resolve(event.name().as_ref());

                if namespace == NS_DAV && local == "prop" {
                    inside_prop = false;
                    prop_depth = 0;
                } else if namespace == NS_DAV && matches!(local.as_str(), "href" | "sync-token") {
                    text_target = None;
                } else if inside_prop && prop_depth > 0 {
                    prop_depth -= 1;
                }
            }
            Event::Eof => break,
            _ => {}
        }

        buf.clear();
    }

    if properties.is_empty() {
        properties.push(PropertyName::dav("getetag"));
    }

    let root = root.ok_or_else(|| AppError::bad_request("REPORT body has no root element"))?;

    let kind = match (root.namespace.as_str(), root.local_name.as_str()) {
        (NS_CALDAV, "calendar-query") => ReportKind::CalendarQuery,
        (NS_CALDAV, "calendar-multiget") => ReportKind::CalendarMultiget,
        (NS_CARDDAV, "addressbook-query") => ReportKind::AddressbookQuery,
        (NS_CARDDAV, "addressbook-multiget") => ReportKind::AddressbookMultiget,
        (NS_DAV, "sync-collection") => ReportKind::SyncCollection,
        _ => return Err(AppError::forbidden("Unsupported REPORT type")),
    };

    Ok(ParsedReport {
        kind,
        properties,
        hrefs,
        sync_token,
    })
}

fn parse_proppatch_body(body: &[u8]) -> Result<Vec<PropertyPatch>, AppError> {
    if body.is_empty() {
        return Ok(Vec::new());
    }

    let mut reader = Reader::from_reader(Cursor::new(body));
    reader.config_mut().trim_text(true);

    let mut buf = Vec::new();
    let mut namespaces = NamespaceScope::default();
    let mut patches = Vec::new();
    let mut inside_prop = false;
    let mut prop_depth = 0usize;
    let mut current_property: Option<PropertyName> = None;
    let mut current_value = String::new();

    loop {
        match reader
            .read_event_into(&mut buf)
            .map_err(|error| AppError::bad_request(format!("Invalid XML: {error}")))?
        {
            Event::Start(event) => {
                namespaces.capture(&event);
                let (namespace, local) = namespaces.resolve(event.name().as_ref());

                if namespace == NS_DAV && local == "prop" {
                    inside_prop = true;
                    prop_depth = 0;
                } else if inside_prop {
                    if prop_depth == 0 {
                        current_property = Some(PropertyName::new(namespace, local));
                        current_value.clear();
                    }
                    prop_depth += 1;
                }
            }
            Event::Empty(event) => {
                namespaces.capture(&event);
                let (namespace, local) = namespaces.resolve(event.name().as_ref());

                if inside_prop && prop_depth == 0 {
                    patches.push(PropertyPatch {
                        property: PropertyName::new(namespace, local),
                        value: None,
                    });
                }
            }
            Event::Text(event) => {
                if current_property.is_some() {
                    current_value.push_str(&decode_xml_text(&event)?);
                }
            }
            Event::End(event) => {
                let (namespace, local) = namespaces.resolve(event.name().as_ref());

                if namespace == NS_DAV && local == "prop" {
                    inside_prop = false;
                    prop_depth = 0;
                } else if inside_prop && prop_depth == 1 {
                    if let Some(property) = current_property.take() {
                        patches.push(PropertyPatch {
                            property,
                            value: Some(current_value.clone()),
                        });
                    }
                    prop_depth = 0;
                } else if inside_prop && prop_depth > 0 {
                    prop_depth -= 1;
                }
            }
            Event::Eof => break,
            _ => {}
        }

        buf.clear();
    }

    Ok(patches)
}

fn parse_collection_body(body: &[u8]) -> Result<ParsedCollectionBody, AppError> {
    if body.is_empty() {
        return Ok(ParsedCollectionBody::default());
    }

    let patches = parse_proppatch_body(body)?;
    let mut parsed = ParsedCollectionBody::default();

    for patch in patches {
        let value = patch.value.unwrap_or_default();

        match (patch.property.namespace.as_str(), patch.property.local_name.as_str()) {
            (NS_DAV, "displayname") => parsed.display_name = Some(value),
            (NS_CALDAV, "calendar-description") => parsed.calendar_description = Some(value),
            (NS_CALDAV, "calendar-timezone") => parsed.calendar_timezone = Some(value),
            (NS_APPLE_ICAL, "calendar-color") => parsed.calendar_color = Some(value),
            (NS_APPLE_ICAL, "calendar-order") => {
                parsed.calendar_order = value.parse::<i32>().ok();
            }
            _ => {}
        }
    }

    parsed.addressbook = body
        .windows(b"addressbook".len())
        .any(|window| window == b"addressbook");

    parsed.supported_calendar_components = vec!["VEVENT".to_owned(), "VTODO".to_owned()];

    Ok(parsed)
}

fn build_multistatus_xml(multistatus: &Multistatus) -> Result<String, AppError> {
    let mut writer = Writer::new(Vec::new());

    writer
        .write_event(Event::Decl(BytesDecl::new("1.0", Some("utf-8"), None)))
        .map_err(xml_write_error)?;

    let mut root = BytesStart::new("d:multistatus");
    root.push_attribute(("xmlns:d", NS_DAV));
    root.push_attribute(("xmlns:c", NS_CALDAV));
    root.push_attribute(("xmlns:card", NS_CARDDAV));
    root.push_attribute(("xmlns:cs", NS_CALENDAR_SERVER));
    root.push_attribute(("xmlns:ical", NS_APPLE_ICAL));
    writer
        .write_event(Event::Start(root))
        .map_err(xml_write_error)?;

    for response in &multistatus.responses {
        write_response(&mut writer, response)?;
    }

    if let Some(sync_token) = multistatus.sync_token.as_deref() {
        write_text_element(&mut writer, "d:sync-token", sync_token)?;
    }

    writer
        .write_event(Event::End(BytesEnd::new("d:multistatus")))
        .map_err(xml_write_error)?;

    String::from_utf8(writer.into_inner()).map_err(|_| AppError::Internal)
}

pub fn build_error_xml(local_name: &str) -> Result<String, AppError> {
    let mut writer = Writer::new(Vec::new());

    writer
        .write_event(Event::Decl(BytesDecl::new("1.0", Some("utf-8"), None)))
        .map_err(xml_write_error)?;

    let mut root = BytesStart::new("d:error");
    root.push_attribute(("xmlns:d", NS_DAV));
    writer
        .write_event(Event::Start(root))
        .map_err(xml_write_error)?;

    let name = format!("d:{local_name}");
    writer
        .write_event(Event::Empty(BytesStart::new(name.as_str())))
        .map_err(xml_write_error)?;

    writer
        .write_event(Event::End(BytesEnd::new("d:error")))
        .map_err(xml_write_error)?;

    String::from_utf8(writer.into_inner()).map_err(|_| AppError::Internal)
}

fn write_response(writer: &mut Writer<Vec<u8>>, response: &ResponseItem) -> Result<(), AppError> {
    writer
        .write_event(Event::Start(BytesStart::new("d:response")))
        .map_err(xml_write_error)?;

    write_text_element(writer, "d:href", &response.href)?;

    if let Some(status) = response.status {
        write_text_element(writer, "d:status", &status_line(status))?;
    }

    for propstat in &response.propstats {
        writer
            .write_event(Event::Start(BytesStart::new("d:propstat")))
            .map_err(xml_write_error)?;

        writer
            .write_event(Event::Start(BytesStart::new("d:prop")))
            .map_err(xml_write_error)?;

        for property in &propstat.properties {
            write_property(writer, property)?;
        }

        writer
            .write_event(Event::End(BytesEnd::new("d:prop")))
            .map_err(xml_write_error)?;

        write_text_element(writer, "d:status", &status_line(propstat.status))?;

        writer
            .write_event(Event::End(BytesEnd::new("d:propstat")))
            .map_err(xml_write_error)?;
    }

    writer
        .write_event(Event::End(BytesEnd::new("d:response")))
        .map_err(xml_write_error)?;

    Ok(())
}

fn write_property(writer: &mut Writer<Vec<u8>>, property: &PropertyValue) -> Result<(), AppError> {
    match &property.value {
        PropertyRenderedValue::Empty => {
            let name = prefixed_name(&property.name);
            writer
                .write_event(Event::Empty(BytesStart::new(name.as_str())))
                .map_err(xml_write_error)?;
        }
        PropertyRenderedValue::Text(value) => {
            let name = prefixed_name(&property.name);
            write_text_element(writer, &name, value)?;
        }
        PropertyRenderedValue::Href(href) => {

#[derive(Default)]
struct NamespaceScope {
    namespaces: HashMap<String, String>,
}

impl NamespaceScope {
    fn capture(&mut self, event: &BytesStart<'_>) {
        for attribute in event.attributes().with_checks(false).flatten() {
            let key = String::from_utf8_lossy(attribute.key.as_ref()).to_string();
            let value = String::from_utf8_lossy(attribute.value.as_ref()).to_string();

            if key == "xmlns" {
                self.namespaces.insert(String::new(), value);
            } else if let Some(prefix) = key.strip_prefix("xmlns:") {
                self.namespaces.insert(prefix.to_owned(), value);
            }
        }
    }

    fn resolve(&self, qname: &[u8]) -> (String, String) {
        let raw = String::from_utf8_lossy(qname).to_string();

        if let Some((prefix, local)) = raw.split_once(':') {
            (
                self.namespaces
                    .get(prefix)
                    .cloned()
                    .unwrap_or_else(|| namespace_from_prefix(prefix).to_owned()),
                local.to_owned(),
            )
        } else {
            (
                self.namespaces
                    .get("")
                    .cloned()
                    .unwrap_or_else(|| NS_DAV.to_owned()),
                raw,
            )
        }
    }
}

pub fn normalize_href(href: &str) -> String {
    let path = href.split('?').next().unwrap_or(href);
    let mut normalized = if path.starts_with('/') {
        path.to_owned()
    } else {
        format!("/{path}")
    };

    while normalized.contains("//") {
        normalized = normalized.replace("//", "/");
    }

    if normalized == "/dav" {
        normalized = "/dav/".to_owned();
    }

    normalized
}

pub fn normalize_collection_href(href: &str) -> String {
    let mut normalized = normalize_href(href);

    if !normalized.ends_with('/') {
        normalized.push('/');
    }

    normalized
}

pub fn split_parent_href_and_name(
    href: &str,
    collection: bool,
) -> Result<(String, String, String), AppError> {
    let normalized = if collection {
        normalize_collection_href(href)
    } else {
        normalize_href(href)
    };

    let trimmed = normalized.trim_end_matches('/');

    let Some(index) = trimmed.rfind('/') else {
        return Err(AppError::conflict("Invalid DAV path"));
    };

    let name = trimmed[index + 1..].to_owned();

    if name.is_empty() {
        return Err(AppError::conflict("Invalid empty resource name"));
    }

    let parent = format!("{}/", &trimmed[..index]);

    Ok((normalized, parent, name))
}

    lower == "text/calendar"
        || lower.starts_with("text/calendar;")
        || lower == "text/vcard"
        || lower.starts_with("text/vcard;")
}

    } else if lower.starts_with("text/vcard") {

    }
```
```rust
}

async fn response_text(response: axum::response::Response) -> String {
    let bytes = to_bytes(response.into_body(), usize::MAX)

        .expect("response body should be readable");

fn assert_xml_has_text(xml: &str, local_name: &str, expected: &str) {
    let mut reader = Reader::from_str(xml);
    reader.config_mut().trim_text(true);

    let mut buf = Vec::new();
    let mut in_target = false;

    loop {
        match reader.read_event_into(&mut buf).expect("valid XML") {
            Event::Start(event) => {
                let name = String::from_utf8_lossy(event.name().as_ref()).to_string();
                in_target = name.ends_with(&format!(":{local_name}")) || name == local_name;
            }
            Event::Text(text) if in_target => {
                let actual = text.unescape().expect("valid XML text").into_owned();
                if actual == expected {
                    return;
                }
                in_target = false;
            }
            Event::End(_) => {
                in_target = false;
            }
            Event::Eof => break,
            _ => {}
        }

        buf.clear();
    }

    panic!("expected XML element {local_name} to contain {expected}");
}
```
