-- 1. Add dtstart and dtend to caldav.calendar_events
ALTER TABLE IF EXISTS caldav.calendar_events
    ADD COLUMN IF NOT EXISTS dtstart TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    ADD COLUMN IF NOT EXISTS dtend TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP;

-- Backfill dtstart and dtend using existing start_time and end_time
DO $$
BEGIN
    IF to_regclass('caldav.calendar_events') IS NOT NULL THEN
        UPDATE caldav.calendar_events
        SET dtstart = start_time,
            dtend = end_time
        WHERE dtstart = CURRENT_TIMESTAMP OR dtstart IS NULL;
    END IF;
END $$;

-- Indexes for CalDAV time-range REPORT queries (RFC 4791)
CREATE INDEX IF NOT EXISTS idx_calendar_events_dtstart_dtend 
    ON caldav.calendar_events(dtstart, dtend);

CREATE INDEX IF NOT EXISTS idx_calendar_events_calendar_dtstart_dtend 
    ON caldav.calendar_events(calendar_id, dtstart, dtend);

-- 2. Add resource_name to carddav.contacts to support URI != UID differences on PUT
ALTER TABLE IF EXISTS carddav.contacts
    ADD COLUMN IF NOT EXISTS resource_name TEXT DEFAULT ('contact-' || gen_random_uuid()::text || '.vcf');

-- Safely backfill resource_name using valid URI sanitization
DO $$
BEGIN
    IF to_regclass('carddav.contacts') IS NOT NULL THEN
        WITH prepared AS (
            SELECT
                id,
                (
                    COALESCE(
                        NULLIF(
                            btrim(
                                regexp_replace(
                                    COALESCE(NULLIF(uid, ''), id::text),
                                    '[^A-Za-z0-9._~-]+',
                                    '-',
                                    'g'
                                ),
                                '.-'
                            ),
                            ''
                        ),
                        'contact-' || substring(id::text, 1, 8)
                    ) || '.vcf'
                ) AS generated_resource_name
            FROM carddav.contacts
            WHERE resource_name IS NULL
               OR resource_name LIKE 'contact-%' AND length(resource_name) > 40
        )
        UPDATE carddav.contacts c
        SET resource_name = prepared.generated_resource_name
        FROM prepared
        WHERE c.id = prepared.id;
    END IF;
END $$;

ALTER TABLE IF EXISTS carddav.contacts
    ALTER COLUMN resource_name SET NOT NULL;

-- Constraints and indexes for precise CardDAV PUT routing
DO $$
BEGIN
    IF to_regclass('carddav.contacts') IS NOT NULL
       AND NOT EXISTS (
           SELECT 1
           FROM pg_constraint
           WHERE conname = 'chk_contacts_resource_name_valid'
             AND conrelid = 'carddav.contacts'::regclass
       )
    THEN
        ALTER TABLE carddav.contacts
            ADD CONSTRAINT chk_contacts_resource_name_valid
            CHECK (
                char_length(resource_name) > 0
                AND position('/' IN resource_name) = 0
                AND position(E'\\' IN resource_name) = 0
                AND resource_name <> '.'
                AND resource_name <> '..'
                AND lower(resource_name) LIKE '%.vcf'
            )
            NOT VALID;
    END IF;
END $$;

CREATE UNIQUE INDEX IF NOT EXISTS idx_contacts_address_book_resource_unique
    ON carddav.contacts(address_book_id, resource_name);

CREATE INDEX IF NOT EXISTS idx_contacts_address_book_resource_etag
    ON carddav.contacts(address_book_id, resource_name, etag);
